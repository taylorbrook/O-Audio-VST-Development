     ```cpp
     float input = data[i];
     float saturated = asymmetricTanh(input, drive, bias);

     float x = std::tanh(input * 0.5f);  // normalize to ~[-1,1]
     float x2 = x * x;
     float h4 = (8.0f * x2 * x2 - 8.0f * x2 + 1.0f) * 0.15f * enhanceAmount;

     saturated += h4;

     float output = input * (1.0f - enhanceAmount) + saturated * enhanceAmount;
     data[i] = std::tanh(output);
     ```

This makes Colored mode noticeably warmer and more intense to match Clean mode's perceived impact.

    ```cpp
    int CleanModeProcessor::getLatencyInSamples() const
    {
        // Safety: Only report latency if we've been properly prepared
        // The oversampler may return garbage if queried before initProcessing()
        if (sampleRate <= 0.0 || maxBlockSize <= 0)
            return 0;

        int latency = 0;

        // Oversampler latency (from harmonic generator)
        int oversamplerLatency = harmonicGenerator.getLatencyInSamples();

        // Sanity check: latency should be positive and reasonable (< 1 second)
        // If oversampler returns garbage, skip it
        if (oversamplerLatency >= 0 && oversamplerLatency < static_cast<int>(sampleRate))
        {
            latency += oversamplerLatency;
        }

        // Lookahead latency (High Fidelity mode only)
        if (currentMode == Mode::HighFidelity && lookaheadSamples > 0)
        {
            latency += lookaheadSamples;
        }

        return latency;
    }
    ```

    This adds three safety guards:
    1. Check that prepare() has been called (sampleRate and maxBlockSize are set)
    2. Validate oversampler latency is non-negative and less than 1 second
    3. Only add lookahead if it's been properly configured

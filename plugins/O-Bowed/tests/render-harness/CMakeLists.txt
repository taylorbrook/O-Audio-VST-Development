# O-Bowed render-harness — Phase 2.1b module-extraction regression baseline
# Mirrors plugins/O-Contrabass/tests/render-harness/CMakeLists.txt verbatim,
# substituting target name / source reach-back paths / JucePlugin_* macros.
#
# Standalone executable. Instantiates OBowedAudioProcessor directly,
# drives a juce::MidiBuffer with the canonical preset (A4 vel 0.7, 5s,
# factory defaults), and writes WAV + JSON status. The bit-exact `cmp` of
# the WAV before vs after R12/R13 module extraction is Gate 2's pass-bar.
#
# Only built when OUARICON_BUILD_TESTS is ON (gated in plugin CMakeLists).

juce_add_console_app(O-Bowed-render-test
    PRODUCT_NAME "O-Bowed-render-test"
)

target_sources(O-Bowed-render-test PRIVATE
    main.cpp

    # Reach back into the plugin source tree — the harness shares the engine
    # implementation, not a separate static library. PluginEditor.cpp is
    # included so the AudioProcessor's createEditor() symbol resolves at link
    # time, even though the harness never calls it.
    #
    # Phase 2.1b: BowModel.cpp + HyperbolicFriction.h are now provided by the
    # bow-friction module via ouaricon_add_module() at the bottom of this file.
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginEditor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/BowedStringVoice.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/WaveguideString.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/BodyResonator.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/SympatheticStringEngine.cpp

    # scala-tuning-engine — required because O-Bowed's PluginProcessor pulls in
    # TuningEngine / ScaleGenerator / EmbeddedTunings / TuningExporter directly.
    ${CMAKE_SOURCE_DIR}/modules/tuning/scala-tuning-engine/cpp/TuningEngine.cpp
    ${CMAKE_SOURCE_DIR}/modules/tuning/scala-tuning-engine/cpp/ScaleGenerator.cpp
    ${CMAKE_SOURCE_DIR}/modules/tuning/scala-tuning-engine/cpp/EmbeddedTunings.cpp
    ${CMAKE_SOURCE_DIR}/modules/tuning/scala-tuning-engine/cpp/TuningExporter.cpp

    # note-expression — required because O-Bowed's PluginProcessor constructs
    # Ouaricon::NoteExpression::VST3Extensions and calls drainAndUpdate().
    # The harness binary doesn't run as a VST3, so VST3-specific overrides in
    # cpp/vst3/ are NOT needed; only the SharedCode .cpp implementation.
    ${CMAKE_SOURCE_DIR}/modules/tuning/note-expression/cpp/NoteExpression.cpp
)

target_include_directories(O-Bowed-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP
    ${CMAKE_SOURCE_DIR}/modules/tuning/scala-tuning-engine/cpp
    ${CMAKE_SOURCE_DIR}/modules/tuning/note-expression/cpp
    ${CMAKE_SOURCE_DIR}/modules/persistence/preset-manager/cpp
)

# JuceHeader.h is generated by the plugin target; depend on it so the
# generation runs before the harness build, then add its include path.
add_dependencies(O-Bowed-render-test O-Bowed)
target_include_directories(O-Bowed-render-test PRIVATE
    $<TARGET_PROPERTY:O-Bowed,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-Bowed-render-test PRIVATE
    # Tell PluginProcessor.cpp it is being built into a console host, not a
    # plugin — disables createPluginFilter() registration we don't need.
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    # O-Bowed's PluginEditor uses juce::WebSliderRelay/WebComboBoxRelay
    # (juce_gui_extra web browser API). Enable JUCE_WEB_BROWSER so the
    # editor's symbol references resolve at link time, even though the
    # harness never instantiates the editor.
    JUCE_WEB_BROWSER=1
    JucePlugin_Build_Standalone=1

    # JucePlugin_* macros normally generated by juce_add_plugin (via JuceHeader.h).
    # The harness reuses PluginProcessor.{h,cpp} directly, so we mirror the minimal
    # set the processor references at compile time.
    JucePlugin_Name="O-Bowed"
    JucePlugin_Desc="O-Bowed"
    JucePlugin_Manufacturer="Ouaricon"
    JucePlugin_ManufacturerCode=0x4f756172
    JucePlugin_PluginCode=0x4f427764
    JucePlugin_VersionString="1.3.0"
    JucePlugin_VersionCode=0x10300
    JucePlugin_IsSynth=1
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-Bowed-render-test
    PRIVATE
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
        # GUI modules required because O-Bowed's PluginEditor.h pulls in
        # WebSliderRelay / WebComboBoxRelay (juce_gui_extra). The editor is
        # never instantiated by the harness, but its symbols must resolve.
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_gui_extra
        # PluginEditor.cpp references BinaryData::* (index_html, botanical_png,
        # tuningpanel_js, etc.) — provided by the plugin target's UIResources.
        O-Bowed_UIResources
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

# Phase 2.1b — bow-friction module pulls HyperbolicFriction.h + BowModel.{h,cpp}
# into the harness binary alongside the per-plugin reach-back sources above.
include(${CMAKE_SOURCE_DIR}/modules/cmake/OuariconModules.cmake)
ouaricon_add_module(O-Bowed-render-test bow-friction)

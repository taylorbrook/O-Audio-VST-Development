# O-simpleSampler render-harness — Stage 2 DSP correctness gate.
#
# Standalone console app. Instantiates OSimpleSamplerAudioProcessor directly,
# drives MIDI, renders offline, and verifies the sampler acceptance criteria:
# makes sound, Repitch tuning (f0 / octave scaling), Stretch pitch-tracks-key,
# Stretch time-INDEPENDENCE (the pitch/time-coupling proof), loop-seam continuity
# (forward + ping-pong), region-end declick, Vintage clean-at-zero (DSP-04),
# anti-alias up-transpose stability, and a stress bound. Off by default; enable
# with -DOUARICON_BUILD_TESTS=ON.
#
# SIMPLER than O-simpleGrain's harness: there is NO O-simpleSampler_UIResources
# target yet (Stage 3) and the editor is the minimal PLACEHOLDER (plain
# juce::AudioProcessorEditor — NO WebView / NO UIBinaryData symbols), so we depend
# on O-simpleSampler + O-simpleSampler_Samples ONLY and disable the web browser.

juce_add_console_app(O-simpleSampler-render-test
    PRODUCT_NAME "O-simpleSampler-render-test"
)

target_sources(O-simpleSampler-render-test PRIVATE
    main.cpp
    # Reach into the plugin source tree — share the engine, not a static lib.
    # PluginEditor.cpp is deliberately NOT compiled here: post-Stage-3 it includes
    # UIBinaryData.h and uses juce::WebBrowserComponent/WebSliderRelay types that are
    # unavailable under JUCE_WEB_BROWSER=0, so it goes un-buildable in this harness
    # (project pattern: render_harness_breaks_on_webview_editor). createEditor() now
    # resolves inside PluginProcessor.cpp via its `#else GenericAudioProcessorEditor`
    # branch (PluginProcessor.cpp createEditor()), so the editor TU is not needed at link time.
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
)

target_include_directories(O-simpleSampler-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h + BinaryData.h are generated by the plugin / samples binary-data
# targets; depend on them, then borrow the plugin's include dirs (which include
# the generated JuceLibraryCode dir). NO O-simpleSampler_UIResources — it does not
# exist until Stage 3, and the placeholder editor references no UIBinaryData::*.
add_dependencies(O-simpleSampler-render-test O-simpleSampler O-simpleSampler_Samples)
target_include_directories(O-simpleSampler-render-test PRIVATE
    $<TARGET_PROPERTY:O-simpleSampler,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-simpleSampler-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    # Pump the message loop (runDispatchLoopUntil) so the AsyncUpdater-driven
    # sourceSample decode/zero-cross snap runs in this console app.
    JUCE_MODAL_LOOPS_PERMITTED=1
    # The placeholder editor is a plain juce::AudioProcessorEditor (NO WebView,
    # unlike O-simpleGrain) — disable the web browser so juce_gui_extra does not
    # pull WebKit into the console-app link.
    JUCE_WEB_BROWSER=0
    JucePlugin_Build_Standalone=1

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h). Dev codes
    # (OuDv / OsSm) per root CMakeLists OUARICON_* + the plugin CMake PLUGIN_CODE.
    JucePlugin_Name="O-simpleSampler"
    JucePlugin_Desc="O-simpleSampler"
    JucePlugin_Manufacturer="Ouaricon Audio Development"
    JucePlugin_ManufacturerCode=0x4f754476    # 'OuDv'
    JucePlugin_PluginCode=0x4f73536d          # 'OsSm'
    JucePlugin_VersionString="0.1.0"
    JucePlugin_VersionCode=0x000100
    JucePlugin_IsSynth=1
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-simpleSampler-render-test
    PRIVATE
        # Embedded source built-ins — supplies BinaryData::piano_wav that the
        # processor decodes at prepareToPlay.
        O-simpleSampler_Samples
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_gui_extra
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

# correctness gate (Phase 2.1 onward).
#
# Template: plugins/O-Emulator/tests/render-harness/CMakeLists.txt (8ec32da9).
# The plugin CMake target is O-simpleAmbisonics — same as the folder name.
# Off by default; enable with -DOUARICON_BUILD_TESTS=ON.

juce_add_console_app(O-simpleAmbisonics-render-test
    PRODUCT_NAME "O-simpleAmbisonics-render-test"
)

target_sources(O-simpleAmbisonics-render-test PRIVATE
    main.cpp

    # Reach into the plugin source tree — share the engine, not a static lib.
    # Source/dsp/*.h are header-only and arrive via the include dir below.
    #
    # The editor TU is deliberately NOT listed here and must never be: this
    # target defines JUCE_WEB_BROWSER=0, and when Stage 3 adds a WebView
    # editor its include in PluginProcessor.cpp must sit inside a
    # #if JUCE_WEB_BROWSER guard so this target keeps compiling
    # (pattern_render_harness_breaks_on_webview_editor).
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
)

target_include_directories(O-simpleAmbisonics-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h is generated by the plugin target; depend on it, then borrow
# the plugin's include dirs so <JuceHeader.h> resolves here.
add_dependencies(O-simpleAmbisonics-render-test O-simpleAmbisonics)
target_include_directories(O-simpleAmbisonics-render-test PRIVATE
    $<TARGET_PROPERTY:O-simpleAmbisonics,INCLUDE_DIRECTORIES>
)

# ── Plugin version, DERIVED — never mirrored ──────────────────────────────────
# juce_add_plugin stores the VERSION it was given in the target's JUCE_VERSION
# property; reading it back reads the single source of truth rather than a
# copy of it (pattern_test_fixture_mirrors_drift_silently).
#
# The hex math mirrors JUCE's own _juce_version_code()
# ((major << 16) + (minor << 8) + patch) rather than calling it: that function
# is private to JUCEUtils.cmake and an upstream rename would break configure.
get_target_property(_OSAM_VERSION O-simpleAmbisonics JUCE_VERSION)

if(NOT _OSAM_VERSION)
    message(FATAL_ERROR
        "Could not read JUCE_VERSION from the O-simpleAmbisonics target. Refusing to stamp the "
        "render harness with a guessed version — that is the drift this block exists to prevent.")
endif()

string(REPLACE "." ";" _OSAM_VERSION_PARTS "${_OSAM_VERSION}")
list(GET _OSAM_VERSION_PARTS 0 _OSAM_VER_MAJOR)
list(GET _OSAM_VERSION_PARTS 1 _OSAM_VER_MINOR)
list(GET _OSAM_VERSION_PARTS 2 _OSAM_VER_PATCH)
math(EXPR _OSAM_VERSION_CODE
    "(${_OSAM_VER_MAJOR} << 16) + (${_OSAM_VER_MINOR} << 8) + ${_OSAM_VER_PATCH}"
    OUTPUT_FORMAT HEXADECIMAL)

message(STATUS
    "O-simpleAmbisonics render-harness: version ${_OSAM_VERSION} (${_OSAM_VERSION_CODE}) derived "
    "from the plugin target")

target_compile_definitions(O-simpleAmbisonics-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    JUCE_WEB_BROWSER=0
    JucePlugin_Build_Standalone=1
    OUARICON_RENDER_HARNESS=1

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h). Dev codes per
    # the root CMakeLists OUARICON_* — OuDv is the manufacturer, OSAm the
    # plugin code.
    JucePlugin_Name="O-simpleAmbisonics"
    JucePlugin_Desc="O-simpleAmbisonics"
    JucePlugin_Manufacturer="Ouaricon Audio Development"
    JucePlugin_ManufacturerCode=0x4f754476
    JucePlugin_PluginCode=0x4f53416d
    JucePlugin_VersionString="${_OSAM_VERSION}"
    JucePlugin_VersionCode=${_OSAM_VERSION_CODE}
    JucePlugin_IsSynth=0
    JucePlugin_WantsMidiInput=0
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

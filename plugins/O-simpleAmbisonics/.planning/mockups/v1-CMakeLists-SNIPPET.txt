# ── 2. WebView UI resources (embedded as BinaryData) ─────────────────────────
# THE ONLY juce_add_binary_data TARGET THIS PLUGIN EVER HAS: the HRTF bank is a
# generated header (Source/dsp/BinauralFilterData.h), never a second target
# (critical_dual_binary_data_namespace_collision). Default namespace BinaryData.
#
# Every file listed here is served by a getResource() branch in
# PluginEditor.cpp. juce_add_binary_data STRIPS hyphens from symbol names:
#   eb-garamond.css -> ebgaramond_css, EBGaramond-Regular.woff2 -> EBGaramondRegular_woff2.
# Place this block AFTER target_sources and BEFORE target_link_libraries.

    SOURCES
        Source/ui/public/index.html
        Source/ui/public/js/i18n.js
        Source/ui/public/js/juce/index.js
        Source/ui/public/js/juce/check_native_interop.js

        ${CMAKE_SOURCE_DIR}/modules/ui/eb-garamond/css/eb-garamond.css
        ${CMAKE_SOURCE_DIR}/modules/ui/eb-garamond/fonts/EBGaramond-Regular.woff2

        ${CMAKE_SOURCE_DIR}/modules/ui/eb-garamond/fonts/EBGaramond-Bold.woff2
)

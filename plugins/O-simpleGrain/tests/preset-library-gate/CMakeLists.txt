# O-simpleGrain v1.9.0 preset/library gate — derived from tests/render-harness.
# Drives the processor's preset + sample-library methods directly (they are
# native functions in the plugin, unreachable from pedalboard), in a temporary
# HOME so the user's real ~/Library/O-simpleGrain is never touched.
# Off by default; built with -DOUARICON_BUILD_TESTS=ON.

juce_add_console_app(O-simpleGrain-library-gate
    PRODUCT_NAME "O-simpleGrain-library-gate"
)

target_sources(O-simpleGrain-library-gate PRIVATE
    main.cpp
    # Reach into the plugin source tree — share the engine, not a static lib.
    # PluginEditor.cpp is included so createEditor() resolves at link time.
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginEditor.cpp
)

target_include_directories(O-simpleGrain-library-gate PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h + BinaryData.h + UIBinaryData.h are generated by the plugin /
# binary-data targets; depend on them, then borrow the plugin's include dirs.
# O-simpleGrain_UIResources (NAMESPACE UIBinaryData) is required because the
# Stage-3 PluginEditor.cpp getResource() references UIBinaryData::* — without it
# the harness link fails on those symbols.
add_dependencies(O-simpleGrain-library-gate O-simpleGrain O-simpleGrain_Samples O-simpleGrain_UIResources)
target_include_directories(O-simpleGrain-library-gate PRIVATE
    $<TARGET_PROPERTY:O-simpleGrain,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-simpleGrain-library-gate PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    # Pump the message loop (runDispatchLoopUntil) so the AsyncUpdater-driven
    # sourceSample decode/hot-swap runs in this console app.
    JUCE_MODAL_LOOPS_PERMITTED=1
    # The processor declares the WebView drag-drop native-fn bridge (Stage 2.3)
    # and the minimal editor compiles here only so createEditor() links; enable
    # the web-browser path so juce_gui_extra references resolve.
    JUCE_WEB_BROWSER=1
    JucePlugin_Build_Standalone=1

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h). Dev codes
    # (OuDv / OsGr) per root CMakeLists OUARICON_*.
    JucePlugin_Name="O-simpleGrain"
    JucePlugin_Desc="O-simpleGrain"
    JucePlugin_Manufacturer="Ouaricon Audio Development"
    JucePlugin_ManufacturerCode=0x4f754476
    JucePlugin_PluginCode=0x4f734772
    # Version macros derive from the plugin CMakeLists' single source of truth
    # (WR-04 — these were hand-rolled and drifted to 1.0.1 while the plugin shipped on).
    JucePlugin_VersionString="${OSIMPLEGRAIN_VERSION}"
    JucePlugin_VersionCode=${OSIMPLEGRAIN_VERSION_CODE}
    JucePlugin_IsSynth=1
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-simpleGrain-library-gate
    PRIVATE
        # Embedded source built-ins — supplies BinaryData::fire_wav etc. that the
        # processor decodes at prepareToPlay / on sourceSample change.
        O-simpleGrain_Samples
        # WebView UI resources — supplies UIBinaryData::index_html etc. that the
        # Stage-3 editor's getResource() serves; needed so PluginEditor.cpp links.
        O-simpleGrain_UIResources
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_gui_extra
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

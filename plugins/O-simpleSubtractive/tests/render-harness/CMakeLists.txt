# O-simpleSubtractive render-harness — Stage 2 DSP correctness gate.
#
# Standalone console app. Instantiates OSimpleSubtractiveAudioProcessor directly,
# drives MIDI, renders offline, and verifies the subtractive engine: makes sound,
# correct pitch, the four filter modes + three slopes, dual-ADSR independence,
# key-tracking, high-key aliasing budget, clean self-oscillation (+ in tune), the
# closed-form magnitude curve vs the measured filter, and Poly/Mono/Legato + glide.
# Off by default; enable with -DOUARICON_BUILD_TESTS=ON.

juce_add_console_app(O-simpleSubtractive-render-test
    PRODUCT_NAME "O-simpleSubtractive-render-test"
)

target_sources(O-simpleSubtractive-render-test PRIVATE
    main.cpp
    # Reach into the plugin source tree — share the engine, not a static lib.
    # PluginEditor.cpp is included so createEditor() resolves at link time.
    # Stage 3: the editor is a juce::WebBrowserComponent UI (WebSliderRelay /
    # WebComboBoxRelay), so the editor translation unit must compile with the
    # web-browser path ENABLED and link O-simpleSubtractive_UIResources (the
    # O-simpleFM working pattern) — with the web browser disabled the WebView
    # types don't exist and the harness fails to build
    # (pattern_render_harness_breaks_on_webview_editor).
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginEditor.cpp
)

target_include_directories(O-simpleSubtractive-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h is generated by the plugin target; depend on it, then borrow its
# include dirs.
add_dependencies(O-simpleSubtractive-render-test O-simpleSubtractive)
target_include_directories(O-simpleSubtractive-render-test PRIVATE
    $<TARGET_PROPERTY:O-simpleSubtractive,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-simpleSubtractive-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    # The Stage-3 editor is a juce::WebBrowserComponent WebView UI. PluginEditor.cpp
    # is compiled here only so createEditor() resolves at link time, so the WebView
    # path MUST be enabled and the embedded UI BinaryData target linked (below).
    JUCE_WEB_BROWSER=1
    JucePlugin_Build_Standalone=1

    # Minimal JucePlugin_* macros the processor references at compile time.
    JucePlugin_Name="O-simpleSubtractive"
    JucePlugin_Desc="O-simpleSubtractive"
    JucePlugin_Manufacturer="Ouaricon"
    JucePlugin_ManufacturerCode=0x4f756172
    JucePlugin_PluginCode=0x4f536953
    JucePlugin_VersionString="1.0.0"
    JucePlugin_VersionCode=0x10000
    JucePlugin_IsSynth=1
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-simpleSubtractive-render-test
    PRIVATE
        # Embedded UI resources — supplies BinaryData::index_html/styles_css/app_js/
        # index_js/check_native_interop_js referenced by PluginEditor.cpp getResource().
        O-simpleSubtractive_UIResources
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_gui_extra
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

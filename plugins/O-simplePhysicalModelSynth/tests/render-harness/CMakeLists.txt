# O-simplePhysicalModelSynth render-harness — Stage-1 build/link smoke-stub.
#
# Standalone console app. Instantiates OSimplePhysicalModelSynthAudioProcessor
# directly, prepares it, pushes a note, renders a few blocks, and asserts the
# output is finite + no crash. At Stage 1 this ONLY proves the harness builds and
# links against the silent shell — locking the JUCE_WEB_BROWSER=0 / no-editor seam
# early. The autocorrelation pitch probe + DSP correctness checks arrive in Stage 2.1.
#
# ⚠ Diverges from O-simpleFM's CURRENT harness (the cautionary post-Stage-3
# retrofit): we do NOT compile the PluginEditor translation unit. createEditor is declared in
# PluginProcessor.h and defined in PluginProcessor.cpp behind #if JUCE_WEB_BROWSER
# (the guarded #include "PluginEditor.h" there vanishes under =0, resolving to a
# GenericAudioProcessorEditor). We set JUCE_WEB_BROWSER=0, and we link NO
# *_UIResources binary-data target. Off by default; enable with -DOUARICON_BUILD_TESTS=ON.

juce_add_console_app(O-simplePhysicalModelSynth-render-test
    PRODUCT_NAME "O-simplePhysicalModelSynth-render-test"
)

target_sources(O-simplePhysicalModelSynth-render-test PRIVATE
    main.cpp
    # Reach into the plugin source tree — share the processor, not a static lib.
    # The voice is header-only; NO PluginEditor translation unit (createEditor's definition in
    # PluginProcessor.cpp resolves to GenericAudioProcessorEditor under JUCE_WEB_BROWSER=0).
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
    # FactoryPresets.cpp MUST link here: PluginProcessor's ctor calls
    # FactoryPresets::build(), so omitting this is an undefined-symbol link error
    # (the #1 footgun for Stage 4 — same class as the WebView/editor seam).
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/FactoryPresets.cpp
)

target_include_directories(O-simplePhysicalModelSynth-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h is generated by the plugin target; depend on it, then borrow its
# include dirs. (INCLUDE_DIRECTORIES carries only include paths — NOT the plugin's
# JUCE_WEB_BROWSER=1 compile def — so this harness's JUCE_WEB_BROWSER=0 below wins.)
add_dependencies(O-simplePhysicalModelSynth-render-test O-simplePhysicalModelSynth)
target_include_directories(O-simplePhysicalModelSynth-render-test PRIVATE
    $<TARGET_PROPERTY:O-simplePhysicalModelSynth,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-simplePhysicalModelSynth-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    # Stage-1 seam: the processor TU compiles with NO editor/WebView types under =0.
    JUCE_WEB_BROWSER=0
    JucePlugin_Build_Standalone=1

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h). Values are cosmetic
    # for a console app EXCEPT the version, which is load-bearing (see below);
    # only PluginCode mirrors the real OsPM code.
    JucePlugin_Name="O-simplePhysicalModelSynth"
    JucePlugin_Desc="O-simplePhysicalModelSynth"
    JucePlugin_Manufacturer="Ouaricon"
    JucePlugin_ManufacturerCode=0x4f756172
    JucePlugin_PluginCode=0x4f73504d
    # MUST track the plugin version (inherited from the parent CMakeLists).
    # The processor ctor runs initializeFactoryPresets(); the module's
    # .factory-version sentinel compares against JucePlugin_VersionString, so a
    # stale literal here makes every harness run rewrite the user's REAL
    # factory-preset directory with the wrong version stamp. Through v1.1.0 this
    # said "1.0.0" while the plugin shipped 1.1.0 — a live, pre-existing drift.
    JucePlugin_VersionString="${OSIMPLEPHYSICALMODELSYNTH_VERSION}"
    JucePlugin_VersionCode=0x10200
    JucePlugin_IsSynth=1
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-simplePhysicalModelSynth-render-test
    PRIVATE
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_gui_extra
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

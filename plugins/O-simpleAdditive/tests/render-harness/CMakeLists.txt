# O-simpleAdditive render-harness — Stage 2 DSP correctness gate + refill profile.
#
# Standalone console app. Instantiates OSimpleAdditiveAudioProcessor directly,
# drives MIDI, renders offline, and (a) asserts the additive->wavetable
# invariants and (b) with --profile, measures the per-voice refillTable() cost
# (STATIC vs MOVING regimes across host block sizes). Off by default; enable with
# -DOUARICON_BUILD_TESTS=ON.

juce_add_console_app(O-simpleAdditive-render-test
    PRODUCT_NAME "O-simpleAdditive-render-test"
)

target_sources(O-simpleAdditive-render-test PRIVATE
    main.cpp
    # Reach into the plugin source tree — share the engine, not a static lib.
    # PluginEditor.cpp is included so createEditor() resolves at link time.
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginEditor.cpp
)

target_include_directories(O-simpleAdditive-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h is generated by the plugin target; depend on it, then borrow its
# include dirs.
add_dependencies(O-simpleAdditive-render-test O-simpleAdditive)
target_include_directories(O-simpleAdditive-render-test PRIVATE
    $<TARGET_PROPERTY:O-simpleAdditive,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-simpleAdditive-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    # The editor is a WebView; PluginEditor.cpp is compiled here only so
    # createEditor() resolves at link time, so the WebView path must be enabled
    # and the embedded UI BinaryData target linked (below).
    JUCE_WEB_BROWSER=1
    JucePlugin_Build_Standalone=1

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h).
    JucePlugin_Name="O-simpleAdditive"
    JucePlugin_Desc="O-simpleAdditive"
    JucePlugin_Manufacturer="Ouaricon"
    JucePlugin_ManufacturerCode=0x4f756172
    JucePlugin_PluginCode=0x4f536941
    JucePlugin_VersionString="1.0.2"
    JucePlugin_VersionCode=0x10002
    JucePlugin_IsSynth=1
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-simpleAdditive-render-test
    PRIVATE
        # Embedded UI resources — supplies BinaryData::* used by PluginEditor.cpp.
        O-simpleAdditive_UIResources
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_gui_extra
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

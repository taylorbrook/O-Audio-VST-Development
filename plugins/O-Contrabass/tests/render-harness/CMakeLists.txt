# O-Contrabass render-harness — Phase 2.1 stability test binary
# RESEARCH §Q1 + PLAN.md Task 7.
#
# Standalone executable. Instantiates OContrabassAudioProcessor directly,
# drives a juce::MidiBuffer, accumulates output, writes WAV + JSON status.
# Pass-conditions per PLAN Task 7 (no NaN/inf, peak ≤ 0 dBFS, block wall-clock
# not exceeding 5× median, RMS over final 1s within 0.5–2.0× RMS over s5–s6).

# Only built when OUARICON_BUILD_TESTS is ON (gated in plugin CMakeLists).

juce_add_console_app(O-Contrabass-render-test
    PRODUCT_NAME "O-Contrabass-render-test"
)

target_sources(O-Contrabass-render-test PRIVATE
    main.cpp

    # Reach back into the plugin source tree — the harness shares the engine
    # implementation, not a separate static library. PluginEditor.cpp is
    # included so the AudioProcessor's createEditor() symbol resolves at link
    # time, even though the harness never calls it.
    #
    # Phase 2.1b: BowModel.cpp + HyperbolicFriction.h are now provided by the
    # bow-friction module via ouaricon_add_module() at the bottom of this file.
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginEditor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/BowedContrabassVoice.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/WaveguideString.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/BodyResonator.cpp
)

target_include_directories(O-Contrabass-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP
    # scala-tuning-engine and note-expression headers — PluginProcessor.cpp
    # may include them transitively via Stage-1 wiring.
    ${CMAKE_SOURCE_DIR}/modules/tuning/scala-tuning-engine/cpp
    ${CMAKE_SOURCE_DIR}/modules/tuning/note-expression/cpp
)

# JuceHeader.h is generated by the plugin target; depend on it so the
# generation runs before the harness build, then add its include path.
add_dependencies(O-Contrabass-render-test O-Contrabass)
target_include_directories(O-Contrabass-render-test PRIVATE
    $<TARGET_PROPERTY:O-Contrabass,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-Contrabass-render-test PRIVATE
    # Tell PluginProcessor.cpp it is being built into a console host, not a
    # plugin — disables createPluginFilter() registration we don't need.
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    JUCE_WEB_BROWSER=0
    # Stub the editor out of this binary — it's audio-only.
    JucePlugin_Build_Standalone=1

    # JucePlugin_* macros normally generated by juce_add_plugin (via JuceHeader.h).
    # The harness reuses PluginProcessor.{h,cpp} directly, so we mirror the minimal
    # set the processor references at compile time.
    JucePlugin_Name="O-Contrabass"
    JucePlugin_Desc="O-Contrabass"
    JucePlugin_Manufacturer="Ouaricon"
    JucePlugin_ManufacturerCode=0x4f756172
    JucePlugin_PluginCode=0x4f436273
    JucePlugin_VersionString="1.0.0"
    JucePlugin_VersionCode=0x10000
    JucePlugin_IsSynth=1
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-Contrabass-render-test
    PRIVATE
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

# Phase 2.1b — bow-friction module pulls HyperbolicFriction.h + BowModel.{h,cpp}
# into the harness binary alongside the per-plugin reach-back sources above.
include(${CMAKE_SOURCE_DIR}/modules/cmake/OuariconModules.cmake)
ouaricon_add_module(O-Contrabass-render-test bow-friction)

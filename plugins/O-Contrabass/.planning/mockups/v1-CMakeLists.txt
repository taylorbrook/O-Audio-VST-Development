# WebView UI resources — embedded as binary data, served by the editor's
# resource provider (bare-path lookups: "/", "/index.html", "/js/juce/*.js").
#
# GOTCHA (MEMORY: critical_dual_binary_data_namespace_collision): if a SECOND
# juce_add_binary_data target is ever added to this plugin (e.g. embedded
# samples), give one of them a distinct NAMESPACE — two targets defaulting to
# `BinaryData` collide at link time with duplicate symbols. O-Contrabass has
# only this one today, so the default BinaryData namespace is fine and matches
# the `#include "BinaryData.h"` in v1-PluginEditor.cpp.
# -----------------------------------------------------------------------------
juce_add_binary_data(OContrabass_UIResources
    SOURCES
        Source/ui/public/index.html
        Source/ui/public/js/juce/index.js
        Source/ui/public/js/juce/check_native_interop.js
)

target_link_libraries(O-Contrabass
    PRIVATE
        OContrabass_UIResources
)

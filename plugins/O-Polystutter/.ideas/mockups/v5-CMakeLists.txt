# WebView UI Resources (HTML, JS, images)
juce_add_binary_data(${PROJECT_NAME}_UIResources
    SOURCES
        Source/ui/public/index.html
        Source/ui/public/js/juce/index.js
        Source/ui/public/js/juce/check_native_interop.js
        Source/ui/public/img/paper-background.jpg
        Source/ui/public/img/botanical-bug.png
)

# Link UI resources and required JUCE modules
target_link_libraries(${PROJECT_NAME}
    PRIVATE
        ${PROJECT_NAME}_UIResources
        juce::juce_gui_extra  # REQUIRED for WebBrowserComponent
        juce::juce_audio_processors
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)

# Generate JuceHeader.h (JUCE 8 requirement - see juce8-critical-patterns.md #1)
juce_generate_juce_header(${PROJECT_NAME})

# WebView compile definitions (REQUIRED - see juce8-critical-patterns.md #3)
target_compile_definitions(${PROJECT_NAME}
    PUBLIC
        JUCE_WEB_BROWSER=1     # Enable WebView support
        JUCE_USE_CURL=0        # Disable CURL (not needed for local HTML)
        JUCE_VST3_CAN_REPLACE_VST2=0
)

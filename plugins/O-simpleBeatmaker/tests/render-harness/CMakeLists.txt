# O-simpleBeatmaker render-harness — Stage 2 DSP correctness gate.
#
# Standalone console app. Instantiates OSimpleBeatmakerAudioProcessor directly,
# injects a synthetic AudioPlayHead, drives MIDI + the internal sequencer, and
# asserts the 6 timing/voice probes (grid accuracy, swing, humanize+quantize,
# block-boundary, MIDI-playable voices, viz truth). Off by default; enable with
# -DOUARICON_BUILD_TESTS=ON.
#
# DIVERGES from the O-simpleFM template: NO *_UIResources link and NO
# JUCE_WEB_BROWSER=1 — this is a headless DSP gate. Since Stage 3 the real editor
# is a WebView (juce::WebBrowserComponent + relays/attachments), so the PluginEditor
# translation unit is NOT compiled here; under JUCE_WEB_BROWSER=0 the processor's createEditor()
# falls back to a GenericAudioProcessorEditor (see PluginProcessor.cpp), which is
# all the linker needs (the harness never opens an editor).

juce_add_console_app(O-simpleBeatmaker-render-test
    PRODUCT_NAME "O-simpleBeatmaker-render-test"
)

target_sources(O-simpleBeatmaker-render-test PRIVATE
    main.cpp
    # Reach into the plugin source tree — share the engine, not a static lib.
    # the PluginEditor translation unit is intentionally excluded (WebView; needs JUCE_WEB_BROWSER=1).
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
)

target_include_directories(O-simpleBeatmaker-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h is generated by the plugin target; depend on it, then borrow its
# include dirs.
add_dependencies(O-simpleBeatmaker-render-test O-simpleBeatmaker)
target_include_directories(O-simpleBeatmaker-render-test PRIVATE
    $<TARGET_PROPERTY:O-simpleBeatmaker,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-simpleBeatmaker-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    JUCE_WEB_BROWSER=0           # Stage-2 editor is a GenericAudioProcessorEditor shell
    JucePlugin_Build_Standalone=1
    OUARICON_BUILD_TESTS=1       # compile in the processor's emitted-MIDI test hook

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h).
    JucePlugin_Name="O-simpleBeatmaker"
    JucePlugin_Desc="O-simpleBeatmaker"
    JucePlugin_Manufacturer="Ouaricon"
    JucePlugin_ManufacturerCode=0x4f756172
    JucePlugin_PluginCode=0x4f536942
    # MUST track the plugin version (inherited from the parent CMakeLists).
    # A stale literal here is a live drift: the processor's own version-stamped
    # behaviour, and on any plugin with a preset manager the .factory-version
    # sentinel, would be built against a version that does not exist. Through
    # v1.0.3 this said "1.0.1" while the plugin shipped 1.0.3 and the code said
    # 0x10000.
    JucePlugin_VersionString="${OSIMPLEBEATMAKER_VERSION}"
    JucePlugin_VersionCode=0x10100
    JucePlugin_IsSynth=1
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-simpleBeatmaker-render-test
    PRIVATE
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_gui_extra
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

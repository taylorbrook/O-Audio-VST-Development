# O-simpleFM render-harness — Stage 2 DSP correctness gate.
#
# Standalone console app. Instantiates OSimpleFMAudioProcessor directly, drives
# MIDI, renders offline, and verifies the PM math: makes sound, correct pitch,
# index -> sidebands, carrier null at I~=2.405 (Bessel J0 zero), feedback
# stability. Off by default; enable with -DOUARICON_BUILD_TESTS=ON.

juce_add_console_app(O-simpleFM-render-test
    PRODUCT_NAME "O-simpleFM-render-test"
)

target_sources(O-simpleFM-render-test PRIVATE
    main.cpp
    # Reach into the plugin source tree — share the engine, not a static lib.
    # PluginEditor.cpp is included so createEditor() resolves at link time.
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginEditor.cpp
)

target_include_directories(O-simpleFM-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h is generated by the plugin target; depend on it, then borrow its
# include dirs.
add_dependencies(O-simpleFM-render-test O-simpleFM)
target_include_directories(O-simpleFM-render-test PRIVATE
    $<TARGET_PROPERTY:O-simpleFM,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-simpleFM-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    JUCE_WEB_BROWSER=0
    JucePlugin_Build_Standalone=1

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h).
    JucePlugin_Name="O-simpleFM"
    JucePlugin_Desc="O-simpleFM"
    JucePlugin_Manufacturer="Ouaricon"
    JucePlugin_ManufacturerCode=0x4f756172
    JucePlugin_PluginCode=0x4f536946
    JucePlugin_VersionString="1.0.0"
    JucePlugin_VersionCode=0x10000
    JucePlugin_IsSynth=1
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-simpleFM-render-test
    PRIVATE
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_gui_extra
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

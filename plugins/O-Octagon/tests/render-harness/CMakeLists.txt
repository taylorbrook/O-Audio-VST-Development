# O-Octagon render harness — the PLUGIN-level half of Phase 2.1's gate.
#
# Pulled forward from Phase 2.2 by CONTEXT-2.1 D3, because it closes two Stage-1 carry-overs a
# phase early and needs no hardware to do it: unity gain was confirmed by INSPECTION only (issue 3),
# and the F3 3-7-channel hazard was REASONED, not measured (issue 4). This target constructs those
# bus layouts programmatically.
#
# Template: plugins/O-ReverseDelay/tests/render-harness/CMakeLists.txt.
#
# CRITICAL: the plugin CMake target is OuariconOctagon, NOT the folder name O-Octagon
# (memory pattern build_script_target_name_vs_folder).
# Off by default; enable with -DOUARICON_BUILD_TESTS=ON.

juce_add_console_app(O-Octagon-render-test
    PRODUCT_NAME "O-Octagon-render-test"
)

target_sources(O-Octagon-render-test PRIVATE
    main.cpp

    # Reach into the plugin source tree — share the engine, not a static lib.
    #
    # The Stage-3 WebView editor is deliberately NOT listed here and must never be: this target
    # defines JUCE_WEB_BROWSER=0, under which WebBrowserComponent's types do not exist.
    # createEditor() is already guarded with #if JUCE_WEB_BROWSER so the swap at Phase 3.1 cannot
    # break this target (pattern_render_harness_breaks_on_webview_editor).
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/Data/VenueModel.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/ChannelMap.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/DSP/ConvexHull2D.cpp
)

target_include_directories(O-Octagon-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h is generated by the plugin target; depend on it, then borrow the plugin's include
# dirs so <JuceHeader.h> resolves here.
add_dependencies(O-Octagon-render-test OuariconOctagon)
target_include_directories(O-Octagon-render-test PRIVATE
    $<TARGET_PROPERTY:OuariconOctagon,INCLUDE_DIRECTORIES>
)

# ── Plugin version, DERIVED — never mirrored (RESEARCH-2.1 G7) ────────────────
# juce_add_plugin stores the VERSION it was given in the target's JUCE_VERSION property, so reading
# it back here reads the single source of truth rather than a copy of it.
#
# The sibling O-ReverseDelay harness documents this literal drifting TWICE across five releases,
# because "keep in sync with" is a promise a comment cannot keep
# (pattern_test_fixture_mirrors_drift_silently). O-Octagon has no factory presets and no version
# sentinels yet, which is exactly why it costs nothing to do correctly now instead of after the
# first thing keys off it.
#
# The hex math mirrors JUCE's own _juce_version_code() ((major << 16) + (minor << 8) + patch)
# rather than calling it: that function is private to JUCEUtils.cmake and an upstream rename would
# break configure.
get_target_property(_OCT_VERSION OuariconOctagon JUCE_VERSION)

if(NOT _OCT_VERSION)
    message(FATAL_ERROR
        "Could not read JUCE_VERSION from the OuariconOctagon target. Refusing to stamp the render "
        "harness with a guessed version — that is the drift this block exists to prevent.")
endif()

string(REPLACE "." ";" _OCT_VERSION_PARTS "${_OCT_VERSION}")
list(GET _OCT_VERSION_PARTS 0 _OCT_VER_MAJOR)
list(GET _OCT_VERSION_PARTS 1 _OCT_VER_MINOR)
list(GET _OCT_VERSION_PARTS 2 _OCT_VER_PATCH)
math(EXPR _OCT_VERSION_CODE
    "(${_OCT_VER_MAJOR} << 16) + (${_OCT_VER_MINOR} << 8) + ${_OCT_VER_PATCH}"
    OUTPUT_FORMAT HEXADECIMAL)

message(STATUS
    "O-Octagon render-harness: version ${_OCT_VERSION} (${_OCT_VERSION_CODE}) derived from the "
    "plugin target")

target_compile_definitions(O-Octagon-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    JUCE_WEB_BROWSER=0
    JucePlugin_Build_Standalone=1

    # Minimal JucePlugin_* macros the processor references at compile time (normally generated by
    # juce_add_plugin via JuceHeader.h). Dev codes per the root CMakeLists OUARICON_* — OuDv is the
    # manufacturer, OuOc the plugin, matching the `aufx OuOc OuDv` triple auval reports.
    JucePlugin_Name="O-Octagon"
    JucePlugin_Desc="O-Octagon"
    JucePlugin_Manufacturer="Ouaricon Audio Development"
    JucePlugin_ManufacturerCode=0x4f754476
    JucePlugin_PluginCode=0x4f754f63
    JucePlugin_VersionString="${_OCT_VERSION}"
    JucePlugin_VersionCode=${_OCT_VERSION_CODE}
    JucePlugin_IsSynth=0
    JucePlugin_WantsMidiInput=0
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

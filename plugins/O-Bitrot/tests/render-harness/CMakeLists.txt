# O-Bitrot render harness — Stage-2 DSP correctness gate (Phase 2.1 onward).
#
# Template: plugins/O-Octagon/tests/render-harness/CMakeLists.txt.
#
# CRITICAL: the plugin CMake target is OBitrot, NOT the folder name O-Bitrot
# (memory pattern build_script_target_name_vs_folder).
# Off by default; enable with -DOUARICON_BUILD_TESTS=ON.

juce_add_console_app(O-Bitrot-render-test
    PRODUCT_NAME "O-Bitrot-render-test"
)

target_sources(O-Bitrot-render-test PRIVATE
    main.cpp

    # Reach into the plugin source tree — share the engine, not a static lib.
    #
    # The editor TU is deliberately NOT listed here and must never be: this
    # target defines JUCE_WEB_BROWSER=0, and createEditor() in
    # PluginProcessor.cpp is guarded with #if JUCE_WEB_BROWSER so the Stage-3
    # WebView swap cannot break this target
    # (pattern_render_harness_breaks_on_webview_editor).
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
)

target_include_directories(O-Bitrot-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h is generated by the plugin target; depend on it, then borrow
# the plugin's include dirs so <JuceHeader.h> resolves here.
add_dependencies(O-Bitrot-render-test OBitrot)
target_include_directories(O-Bitrot-render-test PRIVATE
    $<TARGET_PROPERTY:OBitrot,INCLUDE_DIRECTORIES>
)

# ── Plugin version, DERIVED — never mirrored ──────────────────────────────────
# juce_add_plugin stores the VERSION it was given in the target's JUCE_VERSION
# property; reading it back reads the single source of truth rather than a
# copy of it (pattern_test_fixture_mirrors_drift_silently — the O-ReverseDelay
# harness literal drifted twice across five releases).
#
# The hex math mirrors JUCE's own _juce_version_code()
# ((major << 16) + (minor << 8) + patch) rather than calling it: that function
# is private to JUCEUtils.cmake and an upstream rename would break configure.
get_target_property(_OBR_VERSION OBitrot JUCE_VERSION)

if(NOT _OBR_VERSION)
    message(FATAL_ERROR
        "Could not read JUCE_VERSION from the OBitrot target. Refusing to stamp the render "
        "harness with a guessed version — that is the drift this block exists to prevent.")
endif()

string(REPLACE "." ";" _OBR_VERSION_PARTS "${_OBR_VERSION}")
list(GET _OBR_VERSION_PARTS 0 _OBR_VER_MAJOR)
list(GET _OBR_VERSION_PARTS 1 _OBR_VER_MINOR)
list(GET _OBR_VERSION_PARTS 2 _OBR_VER_PATCH)
math(EXPR _OBR_VERSION_CODE
    "(${_OBR_VER_MAJOR} << 16) + (${_OBR_VER_MINOR} << 8) + ${_OBR_VER_PATCH}"
    OUTPUT_FORMAT HEXADECIMAL)

message(STATUS
    "O-Bitrot render-harness: version ${_OBR_VERSION} (${_OBR_VERSION_CODE}) derived from the "
    "plugin target")

target_compile_definitions(O-Bitrot-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    JUCE_WEB_BROWSER=0
    JucePlugin_Build_Standalone=1

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h). Dev codes per
    # the root CMakeLists OUARICON_* — OuDv is the manufacturer, OBrt the
    # plugin code.
    JucePlugin_Name="O-Bitrot"
    JucePlugin_Desc="O-Bitrot"
    JucePlugin_Manufacturer="Ouaricon Audio Development"
    JucePlugin_ManufacturerCode=0x4f754476
    JucePlugin_PluginCode=0x4f427274
    JucePlugin_VersionString="${_OBR_VERSION}"
    JucePlugin_VersionCode=${_OBR_VERSION_CODE}
    JucePlugin_IsSynth=0
    JucePlugin_WantsMidiInput=0
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

    PRODUCT_NAME "O-Tapestop-render-test"
)

target_sources(O-Tapestop-render-test PRIVATE
    main.cpp

    # Reach into the plugin source tree — share the engine, not a static lib.
    #
    # The Stage-3 WebView editor TU is deliberately NOT listed here and must
    # never be: this target defines JUCE_WEB_BROWSER=0, under which

    # guarded with #if JUCE_WEB_BROWSER (Stage 1), so the Stage-3 swap cannot

    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp

target_include_directories(O-Tapestop-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source

# JuceHeader.h is generated by the plugin target; depend on it, then borrow the
# plugin's include dirs so <JuceHeader.h> resolves here.
add_dependencies(O-Tapestop-render-test OuariconTapestop)
target_include_directories(O-Tapestop-render-test PRIVATE
    $<TARGET_PROPERTY:OuariconTapestop,INCLUDE_DIRECTORIES>
)

# ── Plugin version, DERIVED — never mirrored ──────────────────────────────────
# juce_add_plugin stores the VERSION it was given in the target's JUCE_VERSION
# property, so reading it back here reads the single source of truth rather
# than a copy of it (pattern_test_fixture_mirrors_drift_silently — the
# O-ReverseDelay harness documents this literal drifting TWICE across five
# releases).
#
# The hex math mirrors JUCE's own _juce_version_code()
# ((major << 16) + (minor << 8) + patch) rather than calling it: that function
# is private to JUCEUtils.cmake and an upstream rename would break configure.
get_target_property(_OTSP_VERSION OuariconTapestop JUCE_VERSION)

if(NOT _OTSP_VERSION)
    message(FATAL_ERROR
        "Could not read JUCE_VERSION from the OuariconTapestop target. Refusing to stamp the "
        "render harness with a guessed version — that is the drift this block exists to prevent.")
endif()

string(REPLACE "." ";" _OTSP_VERSION_PARTS "${_OTSP_VERSION}")
list(GET _OTSP_VERSION_PARTS 0 _OTSP_VER_MAJOR)
list(GET _OTSP_VERSION_PARTS 1 _OTSP_VER_MINOR)
list(GET _OTSP_VERSION_PARTS 2 _OTSP_VER_PATCH)
math(EXPR _OTSP_VERSION_CODE
    "(${_OTSP_VER_MAJOR} << 16) + (${_OTSP_VER_MINOR} << 8) + ${_OTSP_VER_PATCH}"
    OUTPUT_FORMAT HEXADECIMAL)

message(STATUS
    "O-Tapestop render-harness: version ${_OTSP_VERSION} (${_OTSP_VERSION_CODE}) derived from "
    "the plugin target")

target_compile_definitions(O-Tapestop-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    JUCE_WEB_BROWSER=0
    JucePlugin_Build_Standalone=1

    # Harness-only instrumentation hook (O-ReverseDelay precedent): exposes
    # TapestopProcessor::getDebtSamplesForTest() for probe P4 (Phase 2.3;
    # API stable from 2.1). Never defined by the plugin target.
    OUARICON_RENDER_HARNESS=1

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h). Dev codes per
    # the root CMakeLists OUARICON_* — OuDv is the manufacturer; OTsp is this
    # plugin's PLUGIN_CODE (plugins/O-Tapestop/CMakeLists.txt:8).
    JucePlugin_Name="O-Tapestop"
    JucePlugin_Desc="O-Tapestop"
    JucePlugin_Manufacturer="Ouaricon Audio Development"
    JucePlugin_ManufacturerCode=0x4f754476
    JucePlugin_PluginCode=0x4f547370
    JucePlugin_VersionString="${_OTSP_VERSION}"
    JucePlugin_VersionCode=${_OTSP_VERSION_CODE}
    JucePlugin_IsSynth=0
    JucePlugin_WantsMidiInput=0
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

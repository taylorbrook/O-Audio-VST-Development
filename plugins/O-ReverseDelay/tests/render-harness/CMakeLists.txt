# O-ReverseDelay render-harness — Stage 2 DSP correctness gate.
#
# Standalone console app. Instantiates ReverseDelayProcessor directly, drives
# the EFFECT with generated input buffers (no MIDI), renders offline, and
# asserts the Phase 2.1 acceptance criteria: reversed-ramp direction, impulse
# reverse bloom, click detector, density loudness flatness, feedback=0 single
# generation. Hard exit codes — exit 0 iff every probe passes.
# Off by default; enable with -DOUARICON_BUILD_TESTS=ON.
#
# CRITICAL: the plugin target is OuariconReverseDelay (NOT the folder name
# O-ReverseDelay) — every target reference below uses it
# (memory pattern build_script_target_name_vs_folder).

juce_add_console_app(O-ReverseDelay-render-test
    PRODUCT_NAME "O-ReverseDelay-render-test"
)

target_sources(O-ReverseDelay-render-test PRIVATE
    main.cpp
    # Reach into the plugin source tree — share the engine, not a static lib.
    # Stage 1/2 has no PluginEditor.cpp (createEditor() returns
    # GenericAudioProcessorEditor from within PluginProcessor.cpp).
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp
)

target_include_directories(O-ReverseDelay-render-test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../Source
)

# JuceHeader.h is generated by the plugin target; depend on it, then borrow
# the plugin's include dirs so <JuceHeader.h> resolves here.
add_dependencies(O-ReverseDelay-render-test OuariconReverseDelay)
target_include_directories(O-ReverseDelay-render-test PRIVATE
    $<TARGET_PROPERTY:OuariconReverseDelay,INCLUDE_DIRECTORIES>
)

target_compile_definitions(O-ReverseDelay-render-test PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0
    JUCE_WEB_BROWSER=0
    JucePlugin_Build_Standalone=1

    # Minimal JucePlugin_* macros the processor references at compile time
    # (normally generated by juce_add_plugin via JuceHeader.h). Dev codes
    # (OuDv / ORvD) per root CMakeLists OUARICON_*.
    JucePlugin_Name="O-ReverseDelay"
    JucePlugin_Desc="O-ReverseDelay"
    JucePlugin_Manufacturer="Ouaricon Audio Development"
    JucePlugin_ManufacturerCode=0x4f754476
    JucePlugin_PluginCode=0x4f527644
    # Keep in sync with the plugin CMakeLists' VERSION (1.0.0). 0x10000 = 65536
    # matches the Stage-1 verified AU version (WR-04: do not let these drift).
    JucePlugin_VersionString="1.0.0"
    JucePlugin_VersionCode=0x10000
    JucePlugin_IsSynth=0
    JucePlugin_WantsMidiInput=0
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
    JucePlugin_EditorRequiresKeyboardFocus=0
)

target_link_libraries(O-ReverseDelay-render-test
    PRIVATE
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_core
        juce::juce_data_structures
        juce::juce_dsp
        juce::juce_events
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_gui_extra
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

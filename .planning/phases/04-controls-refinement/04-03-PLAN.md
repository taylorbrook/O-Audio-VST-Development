   ```cpp
   // Thread-safe limit indicator for UI (0.0 = no limiting, 1.0 = heavy limiting)
   std::atomic<float> limitIndicator { 0.0f };
   juce::SmoothedValue<float> limitIndicatorSmooth;
   ```

3. In PluginProcessor.h, add public accessor:
   ```cpp
   // Limit indicator for UI - returns 0.0 (no limiting) to 1.0 (heavy limiting)
   float getLimitIndicator() const { return limitIndicator.load(); }
   ```

4. In PluginProcessor.cpp prepareToPlay(), initialize limit indicator smoother:
   ```cpp
   // Initialize limit indicator with 100ms decay for smooth UI display
   limitIndicatorSmooth.reset(sampleRate, 0.100);
   limitIndicatorSmooth.setCurrentAndTargetValue(0.0f);
   ```

5. In PluginProcessor.cpp processBlock(), track limiting activity.

   The tanh soft limit at the end of both CleanModeProcessor and the output stage triggers limiting.
   Track when samples approach the -2dB ceiling (0.8):

   After recombineBands() but before output gain section, add:
   ```cpp
   // Track limiting activity for UI indicator
   float maxLimitAmount = 0.0f;
   const float ceiling = 0.8f;  // -2dB auto-limit ceiling

   for (int ch = 0; ch < buffer.getNumChannels(); ++ch)
   {
       const float* data = buffer.getReadPointer(ch);
       for (int i = 0; i < numSamples; ++i)
       {
           float absLevel = std::abs(data[i]);
           if (absLevel > ceiling)
           {
               float limitAmount = (absLevel - ceiling) / ceiling;
               maxLimitAmount = std::max(maxLimitAmount, limitAmount);
           }
       }
   }

   // Update limit indicator (decay handled by SmoothedValue)
   if (maxLimitAmount > 0.0f)
       limitIndicatorSmooth.setTargetValue(juce::jmin(1.0f, maxLimitAmount));
   else if (limitIndicatorSmooth.getTargetValue() > 0.0f)
       limitIndicatorSmooth.setTargetValue(0.0f);

   // Update atomic for UI (skip processing to current value)
   limitIndicatorSmooth.skip(numSamples);
   limitIndicator.store(limitIndicatorSmooth.getCurrentValue());
   ```

This provides a 0-1 value indicating how much the auto-limiter is engaged, with 100ms decay for smooth UI display.
